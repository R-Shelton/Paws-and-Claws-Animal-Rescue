# robots.txt for Paws and Claws Animal Rescue
#
# Tells search engine crawlers which paths on this site they may request.
# Every page of the site is open; only project files that are not part of
# the website itself are closed off.
#
# Note: this file controls crawling, not indexing. A blocked URL can still
# be listed in search results if another site links to it. To keep a page
# out of the index, put <meta name="robots" content="noindex"> in that
# page's <head> instead.

User-agent: *

# Open to every crawler: index.html, animals.html, about.html,
# contact.html, donate.html, and the per-animal pages that the Adopt page
# links to (animaldetail.html?id=1 and so on).
Allow: /

# Project documentation for the repository, not a page of the website.
Disallow: /README.md

# CSS/, JS/ and images/ are deliberately left crawlable. Search engines
# load a page's stylesheet and scripts and render it before judging it, so
# blocking those would make the site look broken to the crawler. That
# matters here because the animal listings are built at runtime by
# JS/animals-data.js and JS/animaldetail.js, and because the photos can
# earn traffic through image search.

# sitemap.xml lists every page of this site for crawlers. The directive takes a full absolute URL, so it
# cannot be written as a relative path.
# Sitemap: https://r-shelton.github.io/Paws-and-Claws-Animal-Rescue/sitemap.xml
